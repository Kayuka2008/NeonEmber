        public void Start(int weapon, int difficulty)
        {
            Mode = "play"; Difficulty = difficulty == 1 ? 1 : 0;
            X = 640; Y = 380; Hp = MaxHp = 100; Time = Xp = 0; NextXp = 18;
            Level = Stage = 1; Kills = Coins = TotalUpgrades = BestCombo = 0; Rerolls = 3;
            DashCooldown = DashTime = Invulnerable = 0; AimX = dashX = 1; AimY = dashY = 0;
            Enemies.Clear(); Shots.Clear(); Orbs.Clear(); Effects.Clear(); Choices.Clear();
            ShopOffers.Clear(); Events.Clear(); Ranks.Clear();
            Ranks[weapon == 1 ? "arc" : weapon == 2 ? "blade" : "bolt"] = 1;
            spawnClock = 0.6f; boltClock = arcClock = bladeClock = 0;
            bladeAngle = phaseClock = idleXpClock = 0; nextShop = 90; nextBoss = 180;
            nextId = pendingLevels = bossCycle = 0; lastPhase = -1; pendingShop = false;
            ResetTactics(); ResetWorld();
            Phase = "蓄势"; Say("无尽协议启动 · 保持移动", 4);
        }

        public void Step(float dt, float mx, float my, bool dash)
        {
            if (Mode != "play") return;
            if (Hp <= 0) { Die(); return; }
            if (float.IsNaN(dt) || float.IsInfinity(dt) || dt <= 0) return;
            dt = Math.Min(dt, 0.05f);
            if (float.IsNaN(mx) || float.IsInfinity(mx)) mx = 0;
            if (float.IsNaN(my) || float.IsInfinity(my)) my = 0;
            ResolveXp();
            if (pendingLevels > 0) { ContinueState(); return; }
            Time += dt; Stage = 1 + (int)(Time / 90);
            DashCooldown = Math.Max(0, DashCooldown - dt);
            Invulnerable = Math.Max(0, Invulnerable - dt);
            AnnouncementTime = Math.Max(0, AnnouncementTime - dt);
            float length = (float)Math.Sqrt(mx * mx + my * my);
            if (length > 1) { mx /= length; my /= length; }
            if (length > 0.05f) { dashX = mx / Math.Max(0.001f, length > 1 ? 1 : length); dashY = my / Math.Max(0.001f, length > 1 ? 1 : length); }
            float previousX = X, previousY = Y;
            bool dashedNow = dash && DashCooldown <= 0;
            if (dashedNow)
            {
                DashTime = 0.20f; DashCooldown = Math.Max(1.5f, 4.4f - Rank("dash") * 0.42f);
                Invulnerable = Math.Max(Invulnerable, 0.30f); Event("dash");
                AddEffect("dash", X, Y, X - dashX * 40, Y - dashY * 40, 24, 0.3f);
            }
            if (DashTime > 0)
            {
                float dashStep = Math.Min(DashTime, dt); DashTime = Math.Max(0, DashTime - dt);
                X += dashX * 760 * dashStep; Y += dashY * 760 * dashStep;
            }
            else { float speed = 220 + Rank("speed") * 16; X += mx * speed * dt; Y += my * speed * dt; }
            X = Clamp(X, Left + 13, Right - 13); Y = Clamp(Y, Top + 13, Bottom - 13);
            ObserveMovement(dt, previousX, previousY);
            UpdateWorld(dt, dashedNow);
            UpdateDirector(dt);
            UpdateTactics(dt);
            UpdateEnemies(dt);
            UpdateHazards(dt);
            if (Mode == "over") return;
            BuildGrid();
            UpdateWeapons(dt);
            UpdateShots(dt);
            if (Mode == "over") return;
            CollectDead();
            UpdateOrbs(dt);
            for (int i = Effects.Count - 1; i >= 0; --i) { Effects[i].Life -= dt; if (Effects[i].Life <= 0) RemoveFast(Effects, i); }
            // Ambient trickle prevents an unlucky scattered drop field from creating a dead progression period.
            idleXpClock += dt;
            if (idleXpClock >= 12) { idleXpClock -= 12; GainXp(2 + Time / 120); }
            ContinueState();
        }

        private void UpdateDirector(float dt)
        {
            phaseClock = Time % 30;
            int phase = phaseClock < 18 ? 0 : phaseClock < 25 ? 1 : 2;
            Phase = phase == 0 ? "游猎" : phase == 1 ? "怪潮" : "喘息";
            if (phase != lastPhase)
            {
                lastPhase = phase;
                if (phase == 1) { Say("怪潮来袭 · 寻找突破口", 2.8f); Event("wave"); }
                else if (phase == 2) { Say("短暂喘息 · 回收经验", 2.5f); }
            }
            if (Time >= nextShop) { nextShop += 90; pendingShop = true; }
            if (Time >= nextBoss)
            {
                nextBoss += 180; bossCycle++;
                bool exists = false;
                for (int i = 0; i < Enemies.Count; i++) if (Enemies[i].Kind == "boss" && Enemies[i].Hp > 0) exists = true;
                if (!exists) { Spawn("boss"); Say("警报 · 监察者降临", 4); Event("boss"); }
            }
            float pressure = 1 + Math.Min(Time / 125, 5.8f);
            float phaseRate = phase == 0 ? 1.1f : phase == 1 ? 2.8f : 0.20f;
            float rate = pressure * phaseRate * (Difficulty == 1 ? 1.25f : 1);
            if (Time >= nextSquad && phase != 2)
            {
                nextSquad = Time + (Time >= 180 ? 21 : 30); SpawnSquad();
                Say("中继编队 · 击破信标可打断合围", 3.5f); Event("wave");
            }
            if (Time >= nextElite && phase != 2)
            {
                nextElite = Time + (Time >= 180 ? 24 : 40);
                int elites = 0;
                for (int i = 0; i < Enemies.Count; i++) if (Enemies[i].Elite && Enemies[i].Hp > 0) elites++;
                if (elites < 8) { Spawn("prism"); if (Time >= 90) Spawn("repairer"); }
            }
            spawnClock -= dt;
            int budget = 6;
            while (spawnClock <= 0 && budget-- > 0)
            {
                spawnClock += 1 / rate;
                double roll = random.NextDouble();
                string kind = "chaser";
                if (Time > 18 && roll < .18) kind = "runner";
                else if (Time > 35 && roll < .34) kind = "shooter";
                else if (Time > 55 && roll < .46) kind = "bomber";
                else if (Time > 25 && roll > .89) kind = "tank";
                Spawn(kind);
            }
        }

        private void Spawn(string kind)
        {
            if (Enemies.Count >= EnemyLimit)
            {
                if (kind != "boss") return;
                int remove = -1;
                for (int i = 0; i < Enemies.Count; i++) if (Enemies[i].Kind != "boss") { remove = i; break; }
                if (remove < 0) return;
                RemoveFast(Enemies, remove);
            }
            float sx = 40, sy = 120;
            for (int tries = 0; tries < 12; tries++)
            {
                int edge = random.Next(4);
                sx = edge == 0 ? Left + 18 : edge == 1 ? Right - 18 : Left + 18 + (float)random.NextDouble() * (Right - Left - 36);
                sy = edge == 2 ? Top + 18 : edge == 3 ? Bottom - 18 : Top + 18 + (float)random.NextDouble() * (Bottom - Top - 36);
                if (DistanceSquared(sx, sy, X, Y) > 260 * 260) break;
            }
            // Bounded scaling: late pressure comes from additional roles and attack patterns.
            float scaling = Math.Min(16, (float)Math.Pow(1 + Time / 130, 1.38)) * (Difficulty == 1 ? 1.2f : 1);
            float radius = kind == "tank" ? 22 : kind == "prism" ? 24 : kind == "boss" ? 38 : kind == "relay" ? 19 : kind == "repairer" ? 17 : kind == "runner" ? 10 : kind == "bomber" ? 17 : 13;
            float health = kind == "boss" ? 1600 * (1 + Math.Min(bossCycle, 8) * 0.55f) : kind == "prism" ? 85 : kind == "repairer" ? 48 : kind == "tank" ? 56 : kind == "relay" ? 62 : kind == "bomber" ? 24 : kind == "shooter" ? 19 : kind == "runner" ? 12 : 18;
            float speed = kind == "runner" ? 112 : kind == "tank" ? 86 : kind == "prism" ? 112 : kind == "repairer" ? 102 : kind == "boss" ? 60 : kind == "relay" ? 98 : kind == "shooter" ? 88 : kind == "bomber" ? 83 : 78;
            Enemy enemy = new Enemy();
            enemy.Id = ++nextId; enemy.Kind = kind; enemy.X = sx; enemy.Y = sy; enemy.R = radius;
            enemy.Hp = enemy.MaxHp = health * (kind == "boss" ? Math.Min(3.2f, (float)Math.Pow(1 + Time / 600, 0.6)) : scaling);
            enemy.Speed = speed * (1 + Math.Min(Time / 1800, 0.32f));
            enemy.Damage = (kind == "tank" || kind == "boss" || kind == "prism" ? 20 : 12) * (1 + Math.Min(Time / 650, 2.5f)) * (Difficulty == 1 ? 1.2f : 1);
            enemy.Elite = kind == "prism" || kind == "repairer";
            float facing = (float)Math.Atan2(Y - sy, X - sx); enemy.FacingX = (float)Math.Cos(facing); enemy.FacingY = (float)Math.Sin(facing);
            enemy.Cooldown = 0.7f + (float)random.NextDouble() * 1.5f;
            if (enemy.Elite) enemy.Cooldown = .12f;
            enemy.Timer = 0.55f; enemy.AttackPhase = -1;
            Enemies.Add(enemy); AddEffect("spawn", sx, sy, 0, 0, radius + 8, 0.55f);
        }

        private void UpdateEnemies(float dt) { UpdateTacticalEnemies(dt); }

        private void BuildGrid()
        {
            for (int i = 0; i < grid.Length; i++) if (grid[i] != null) grid[i].Clear();
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i]; if (e.Hp <= 0 || e.AttackPhase == -1) continue;
                int cell = Cell(e.X, e.Y); if (grid[cell] == null) grid[cell] = new List<Enemy>(8); grid[cell].Add(e);
            }
        }

        private Enemy Nearest(float x, float y, float radius, HashSet<int> exclude)
        {
            Enemy best = null; float bestDistance = radius * radius;
            Enemy focused = FindEnemy(FocusTargetId);
            if (focused != null && focused.Hp > 0 && focused.AttackPhase != -1 && (exclude == null || !exclude.Contains(focused.Id)) && DistanceSquared(x, y, focused.X, focused.Y) < bestDistance) return focused;
            int minX = Math.Max(0, (int)((x - radius) / GridSize)), maxX = Math.Min(GridColumns - 1, (int)((x + radius) / GridSize));
            int minY = Math.Max(0, (int)((y - radius) / GridSize)), maxY = Math.Min(GridRows - 1, (int)((y + radius) / GridSize));
            for (int gy = minY; gy <= maxY; gy++) for (int gx = minX; gx <= maxX; gx++)
            {
                List<Enemy> bucket = grid[gy * GridColumns + gx]; if (bucket == null) continue;
                for (int i = 0; i < bucket.Count; i++)
                {
                    Enemy e = bucket[i]; if (e.Hp <= 0 || (exclude != null && exclude.Contains(e.Id))) continue;
                    float distance = DistanceSquared(x, y, e.X, e.Y);
                    if (distance < bestDistance) { bestDistance = distance; best = e; }
                }
            }
            return best;
        }

        private void UpdateWeapons(float dt)
        {
            boltClock -= dt; arcClock -= dt; bladeClock -= dt; bladeAngle += dt * (2.2f + Rank("haste") * .12f);
            Enemy aim = Nearest(X, Y, 700, null);
            if (aim != null)
            {
                float dx = aim.X - X, dy = aim.Y - Y, d = (float)Math.Sqrt(dx * dx + dy * dy);
                AimX = dx / Math.Max(d, .001f); AimY = dy / Math.Max(d, .001f);
            }
            int bolt = Rank("bolt");
            if (bolt > 0 && boltClock <= 0 && aim != null)
            {
                boltClock = .62f / FireRateMultiplier / (1 + bolt * .08f);
                int count = bolt >= 5 ? 5 : bolt >= 3 ? 3 : 1;
                float damage = (20 + bolt * 6) * PowerMultiplier;
                float angle = (float)Math.Atan2(AimY, AimX);
                for (int j = 0; j < count; j++) AddShot(X + AimX * 16, Y + AimY * 16, angle + (j - (count - 1) / 2f) * .115f, 680, damage, false, bolt >= 5 ? 3 : bolt >= 2 ? 1 : 0, bolt >= 5 ? "nova" : "bolt");
                Event("shoot");
            }
            int arc = Rank("arc");
            if (arc > 0 && arcClock <= 0)
            {
                Enemy target = Nearest(X, Y, 330 + arc * 24, null);
                if (target != null)
                {
                    arcClock = 1.2f / FireRateMultiplier; chainHits.Clear();
                    int chains = arc >= 5 ? 9 : 2 + arc;
                    float sx = X, sy = Y;
                    for (int j = 0; j < chains && target != null; j++)
                    {
                        chainHits.Add(target.Id); AddEffect("arc", sx, sy, target.X, target.Y, arc >= 5 ? 5 : 3, .16f);
                        DamageEnemy(target, (19 + arc * 7) * PowerMultiplier * (arc >= 5 ? 1 : (float)Math.Pow(.9, j)), "arc");
                        sx = target.X; sy = target.Y; target = Nearest(sx, sy, 170 + arc * 13, chainHits);
                    }
                    Event("arc");
                }
            }
            int blade = Rank("blade");
            if (blade > 0 && bladeClock <= 0)
            {
                bladeClock = .14f / FireRateMultiplier;
                int count = blade >= 5 ? 6 : 1 + blade;
                float orbit = 63 + blade * 6;
                for (int j = 0; j < count; j++)
                {
                    float angle = bladeAngle + j * (float)Math.PI * 2 / count;
                    float bx = X + (float)Math.Cos(angle) * orbit, by = Y + (float)Math.Sin(angle) * orbit;
                    DamageArea(bx, by, blade >= 5 ? 27 : 20, (8 + blade * 3) * PowerMultiplier, null);
                    if (blade >= 5)
                    {
                        for (int k = Shots.Count - 1; k >= 0; k--)
                            if (Shots[k].Hostile && DistanceSquared(bx, by, Shots[k].X, Shots[k].Y) < 29 * 29) RemoveFast(Shots, k);
                    }
                }
            }
        }

        private void DamageArea(float x, float y, float radius, float damage, Enemy excluded)
        {
            int minX = Math.Max(0, (int)((x - radius - 40) / GridSize)), maxX = Math.Min(GridColumns - 1, (int)((x + radius + 40) / GridSize));
            int minY = Math.Max(0, (int)((y - radius - 40) / GridSize)), maxY = Math.Min(GridRows - 1, (int)((y + radius + 40) / GridSize));
            for (int gy = minY; gy <= maxY; gy++) for (int gx = minX; gx <= maxX; gx++)
            {
                List<Enemy> bucket = grid[gy * GridColumns + gx]; if (bucket == null) continue;
                for (int i = 0; i < bucket.Count; i++)
                {
                    Enemy e = bucket[i]; if (e == excluded || e.Hp <= 0) continue;
                    if (DistanceSquared(x, y, e.X, e.Y) < (radius + e.R) * (radius + e.R)) DamageEnemy(e, damage, "blade", x, y);
                }
            }
        }

        private void AddShot(float x, float y, float angle, float speed, float damage, bool hostile, int pierce, string kind)
        {
            if (Shots.Count >= ShotLimit) return;
            Shot s = new Shot(); s.X = x; s.Y = y; s.VX = (float)Math.Cos(angle) * speed; s.VY = (float)Math.Sin(angle) * speed;
            s.R = hostile ? 6 : kind == "nova" ? 5 : 4; s.Damage = damage; s.Hostile = hostile;
            s.Pierce = pierce; s.Kind = kind; s.Life = hostile ? 6 : 2;
            Shots.Add(s);
        }

        private void UpdateShots(float dt)
        {
            for (int i = Shots.Count - 1; i >= 0; --i)
            {
                Shot s = Shots[i]; s.X += s.VX * dt; s.Y += s.VY * dt; s.Life -= dt;
                bool remove = s.Life <= 0 || s.X < Left - 20 || s.X > Right + 20 || s.Y < Top - 20 || s.Y > Bottom + 20;
                if (!remove && s.Hostile)
                {
                    if (DistanceSquared(s.X, s.Y, X, Y) < (s.R + 11) * (s.R + 11)) { Hurt(s.Damage); remove = true; }
                }
                else if (!remove)
                {
                    int minX = Math.Max(0, (int)((s.X - 45) / GridSize)), maxX = Math.Min(GridColumns - 1, (int)((s.X + 45) / GridSize));
                    int minY = Math.Max(0, (int)((s.Y - 45) / GridSize)), maxY = Math.Min(GridRows - 1, (int)((s.Y + 45) / GridSize));
                    for (int gy = minY; gy <= maxY && !remove; gy++) for (int gx = minX; gx <= maxX && !remove; gx++)
                    {
                        List<Enemy> bucket = grid[gy * GridColumns + gx]; if (bucket == null) continue;
                        for (int j = 0; j < bucket.Count && !remove; j++)
                        {
                            Enemy e = bucket[j]; if (e.Hp <= 0 || (s.HitIds != null && s.HitIds.Contains(e.Id))) continue;
                            if (DistanceSquared(s.X, s.Y, e.X, e.Y) < (s.R + e.R) * (s.R + e.R))
                            {
                                DamageEnemy(e, s.Damage, s.Kind, Math.Abs(s.VX) + Math.Abs(s.VY) > .01f ? e.X - s.VX : X, Math.Abs(s.VX) + Math.Abs(s.VY) > .01f ? e.Y - s.VY : Y);
                                if (s.Pierce <= 0) remove = true;
                                else { s.Pierce--; if (s.HitIds == null) s.HitIds = new HashSet<int>(); s.HitIds.Add(e.Id); }
                            }
                        }
                    }
                }
                if (remove) RemoveFast(Shots, i);
                if (Mode == "over") return;
            }
        }

        private void DamageEnemy(Enemy e, float damage)
        { DamageEnemy(e, damage, "bolt"); }

        private void DamageEnemy(Enemy e, float damage, string weapon)
        { DamageEnemy(e, damage, weapon, X, Y); }

        private void DamageEnemy(Enemy e, float damage, string weapon, float sourceX, float sourceY)
        {
            if (e.Hp <= 0) return;
            damage = ApplyWorldDamage(e, damage, weapon);
            if (random.NextDouble() < Rank("crit") * .065f) damage *= 2;
            if (IsProtected(e)) damage *= .52f;
            damage = ApplyTacticalDamage(e, damage, sourceX, sourceY);
            e.Hp -= damage; AddEffect("hit", e.X, e.Y, 0, 0, e.R * .5f + 5, .10f); Event("hit");
            if (e.Hp <= 0 && e.Kind == "relay") DisruptRelay(e);
        }

        private void CollectDead()
        {
            for (int i = Enemies.Count - 1; i >= 0; --i)
            {
                Enemy e = Enemies[i]; if (e.Hp > 0) continue;
                if (e.Kind == "relay") DisruptRelay(e);
                CancelEnemyWarning(e.Id);
                if (FocusTargetId == e.Id) FocusTargetId = 0;
                Kills++; Coins += e.Kind == "boss" ? 60 : e.Kind == "relay" ? 8 : e.Elite ? 6 : e.Kind == "tank" ? 3 : 1;
                float value = (e.Kind == "boss" ? 42 : e.Kind == "relay" ? 8 : e.Elite ? 5 : e.Kind == "tank" ? 3 : e.Kind == "bomber" ? 2 : 1) * (1 + Time / 110);
                if (e.Kind == "boss")
                {
                    Rerolls = Math.Min(9, Rerolls + 1); Say("监察者击破 · +60 零件 · +1 重掷", 4); Event("bosskill");
                    Hp = Math.Min(MaxHp, Hp + MaxHp * .2f);
                }
                OnWorldEnemyDeath(e);
                AddOrb(e.X, e.Y, value); AddEffect("death", e.X, e.Y, 0, 0, e.R + 15, .3f); RemoveFast(Enemies, i);
            }
        }

        private void AddOrb(float x, float y, float value)
        {
            if (Orbs.Count >= OrbLimit)
            {
                int nearest = 0; float distance = float.MaxValue;
                for (int i = 0; i < Orbs.Count; i++) { float d = DistanceSquared(x, y, Orbs[i].X, Orbs[i].Y); if (d < distance) { distance = d; nearest = i; } }
                Orbs[nearest].Value += value; Orbs[nearest].R = Math.Min(9, 4 + (float)Math.Sqrt(Orbs[nearest].Value) * .6f); return;
            }
            Orb orb = new Orb(); orb.X = x; orb.Y = y; orb.Value = value; Orbs.Add(orb);
        }

        private void UpdateOrbs(float dt)
        {
            float radius = PickupRadius + (Phase == "喘息" ? 90 : 0);
            for (int i = Orbs.Count - 1; i >= 0; --i)
            {
                Orb o = Orbs[i]; float dx = X - o.X, dy = Y - o.Y, d2 = dx * dx + dy * dy;
                if (d2 < 22 * 22) { GainXp(o.Value); RemoveFast(Orbs, i); Event("pickup"); }
                else if (d2 < radius * radius)
                {
                    float d = (float)Math.Sqrt(d2), speed = 340 + (radius - d) * 3;
                    float step = Math.Min(d, speed * dt); o.X += dx / d * step; o.Y += dy / d * step;
                }
            }
        }

        private void Hurt(float damage)
        {
            if (Invulnerable > 0 || Mode != "play") return;
            Hp = Math.Max(0, Hp - AbsorbWorldDamage(damage * (1 - Math.Min(.42f, Rank("guard") * .06f))));
            Invulnerable = Math.Max(Invulnerable, .6f); AddEffect("hurt", X, Y, 0, 0, 30, .24f); Event("hurt");
            if (Hp <= 0) Die();
        }

        private void Die() { Hp = 0; Mode = "over"; Choices.Clear(); pendingLevels = 0; Event("death"); }
        private void GainXp(float value) { Xp += value * (1 + Rank("scholar") * .18f); ResolveXp(); }
        private void ResolveXp()
        {
            if (float.IsNaN(Xp) || float.IsInfinity(Xp)) Xp = 0;
            int budget = 200;
            while (Xp >= NextXp && budget-- > 0)
            {
                Xp -= NextXp; Level++; pendingLevels++;
                float growth = Level - 1, mature = Math.Max(0, growth - 8);
                NextXp = 18 + growth * 7 + mature * mature * 1.5f;
            }
        }

        private void ContinueState()
        {
            if (Mode == "over" || Hp <= 0) { if (Mode != "over") Die(); return; }
            if (pendingLevels > 0)
            {
                Mode = "level"; RollChoices(); Event("level");
            }
            else if (pendingShop && !HasLivingBoss()) { OpenShop(); }
            else Mode = "play";
        }

        public void Choose(int index)
        {
            if (Mode != "level" || index < 0 || index >= Choices.Count || Hp <= 0) { if (Hp <= 0 && Mode != "menu") Die(); return; }
            Apply(Choices[index].Id); pendingLevels = Math.Max(0, pendingLevels - 1); Choices.Clear(); ContinueState();
        }

        public void Reroll()
        {
            if (Mode != "level" || Rerolls <= 0 || Hp <= 0) return;
            Rerolls--; RollChoices(); Event("reroll");
        }

        public void Buy(int index)
        {
            if (Mode != "shop" || Hp <= 0 || index < 0 || index >= ShopOffers.Count) return;
            Upgrade offer = ShopOffers[index]; if (offer.Price < 0 || Coins < offer.Price) return;
            Coins -= offer.Price; Apply(offer.Id); ShopOffers.RemoveAt(index); Event("buy");
        }

        public void LeaveShop()
        {
            if (Mode != "shop") return;
            ShopOffers.Clear(); Say("补给完成 · 继续突围", 2.5f); ContinueState();
        }

        private void OpenShop()
        {
            pendingShop = false; Mode = "shop"; ShopOffers.Clear();
            Upgrade repair = MakeUpgrade("supply"); repair.Price = 0; ShopOffers.Add(repair);
            List<string> ids = AvailableUpgrades();
            for (int i = 0; i < 2 && ids.Count > 0; i++)
            {
                int pick = random.Next(ids.Count); Upgrade offer = MakeUpgrade(ids[pick]);
                offer.Price = 38 + Rank(offer.Id) * 16 + Stage * 3; ShopOffers.Add(offer); ids.RemoveAt(pick);
            }
            Say("补给窗口 · 战场已暂停", 4); Event("shop");
        }

        private void RollChoices()
        {
            Choices.Clear(); List<string> ids = AvailableUpgrades();
            // Every draft keeps a weapon route available until all three routes are evolved.
            List<string> weapons = new List<string>();
            string[] weaponIds = { "bolt", "arc", "blade" };
            for (int i = 0; i < weaponIds.Length; i++)
            {
                string weapon = weaponIds[i]; if (Rank(weapon) >= 5) continue;
                weapons.Add(weapon);
                if (Rank(weapon) > 0) { weapons.Add(weapon); weapons.Add(weapon); }
            }
            if (weapons.Count > 0)
            {
                string picked = weapons[random.Next(weapons.Count)]; Choices.Add(MakeUpgrade(picked)); ids.Remove(picked);
            }
            // Keep a world-linked build decision in each draft, weighted toward prior investment.
            List<string> schoolChoices = new List<string>();
            for (int i = 0; i < ids.Count; i++)
            {
                if (!Lore.IsSchoolUpgrade(ids[i])) continue;
                schoolChoices.Add(ids[i]);
                if (SchoolPoints(Lore.SchoolOf(ids[i])) > 0) { schoolChoices.Add(ids[i]); schoolChoices.Add(ids[i]); }
            }
            if (schoolChoices.Count > 0 && Choices.Count < 3)
            {
                string picked = schoolChoices[random.Next(schoolChoices.Count)]; Choices.Add(MakeUpgrade(picked)); ids.Remove(picked);
            }
            while (Choices.Count < 3 && ids.Count > 0)
            {
                int pick = random.Next(ids.Count); Choices.Add(MakeUpgrade(ids[pick])); ids.RemoveAt(pick);
            }
        }

        private List<string> AvailableUpgrades()
        {
            string[] all = { "bolt", "arc", "blade", "power", "haste", "vitality", "speed", "magnet", "scholar", "crit", "guard", "dash" };
            List<string> result = new List<string>();
            for (int i = 0; i < all.Length; i++) if (Rank(all[i]) < Cap(all[i])) result.Add(all[i]);
            for (int i = 0; i < Lore.UpgradeIds.Length; i++)
            {
                string id = Lore.UpgradeIds[i];
                if (id == "ember_cascade" && Rank("ember_brand") == 0) continue;
                if (id == "echo_trace" && (Rank("bolt") > 0 ? 1 : 0) + (Rank("arc") > 0 ? 1 : 0) + (Rank("blade") > 0 ? 1 : 0) < 2) continue;
                if (Rank(id) < Cap(id)) result.Add(id);
            }
            // At full build, level rewards continue to offer distinct useful infinite-run decisions.
            if (result.Count < 3) result.Add("overclock");
            if (result.Count < 3) result.Add("reserve");
            if (result.Count < 3) result.Add("repair");
            return result;
        }

        private static int Cap(string id)

            if (Lore.IsSchoolUpgrade(id)) return 3;

            if (id == "speed" || id == "scholar" || id == "crit") return 6;
            if (id == "magnet" || id == "dash") return 5;
            return 7;
        }

        private void Apply(string id)
        {
            if (id == "repair" || id == "supply") { Hp = Math.Min(MaxHp, Hp + MaxHp * (id == "supply" ? .25f : .45f)); AddEffect("heal", X, Y, 0, 0, 40, .5f); return; }
            if (id == "overclock") { Ranks[id] = Rank(id) + 1; TotalUpgrades++; return; }
            if (id == "reserve") { MaxHp += 10; Hp = Math.Min(MaxHp, Hp + 20); TotalUpgrades++; return; }
            int rank = Rank(id); if (rank >= Cap(id)) return;
            Ranks[id] = rank + 1; TotalUpgrades++;
            OnWorldUpgrade(id);
            if (id == "vitality") { MaxHp += 20; Hp = Math.Min(MaxHp, Hp + 30); }
            if ((id == "bolt" || id == "arc" || id == "blade") && rank == 4)
            {
                AddEffect("evolve", X, Y, 0, 0, 130, .8f); Say("武器进化 · " + MakeUpgrade(id).Tag, 4); Event("evolve");
            }
            else Event("upgrade");
        }

        private void AddEffect(string kind, float x, float y, float x2, float y2, float radius, float life)
        {
            if (Effects.Count >= EffectLimit) return;
            Effect e = new Effect(); e.Kind = kind; e.X = x; e.Y = y; e.X2 = x2; e.Y2 = y2;
            e.R = radius; e.Life = e.MaxLife = life; Effects.Add(e);
        }
        private void Say(string message, float duration) { Announcement = message; AnnouncementTime = duration; }
        private bool HasLivingBoss() { for (int i = 0; i < Enemies.Count; i++) if (Enemies[i].Kind == "boss" && Enemies[i].Hp > 0) return true; return false; }
        private void Event(string name) { if (Events.Count < 48 && !Events.Contains(name)) Events.Add(name); }
        private static float Clamp(float value, float min, float max) { return Math.Max(min, Math.Min(max, value)); }
        private static float DistanceSquared(float ax, float ay, float bx, float by) { float x = ax - bx, y = ay - by; return x * x + y * y; }
        private static int Cell(float x, float y) { return Math.Max(0, Math.Min(GridRows - 1, (int)(y / GridSize))) * GridColumns + Math.Max(0, Math.Min(GridColumns - 1, (int)(x / GridSize))); }
        private static void RemoveFast<T>(List<T> list, int index) { int last = list.Count - 1; list[index] = list[last]; list.RemoveAt(last); }
    }

        private void ObserveMovement(float dt, float previousX, float previousY)
        {
            // Only displacement already observed is available to enemy planning. Do not use input.
            float vx = (X - previousX) / dt, vy = (Y - previousY) / dt;
            float speed = (float)Math.Sqrt(vx * vx + vy * vy);
            ObservedDashAge = Math.Min(99, ObservedDashAge + dt);
            // A fast displacement is evidence that a dash happened. Never inspect dash input/cooldown.
            if (speed > 440) { ObservedDashAge = 0; turnHistory = Math.Max(0, turnHistory - dt * 2); }
            float previousSpeed = (float)Math.Sqrt(lastObservedVX * lastObservedVX + lastObservedVY * lastObservedVY);
            if (speed > 80 && speed < 440 && previousSpeed > 80 && ObservedDashAge > .22f)
            {
                float turn = (float)Math.Atan2(lastObservedVX * vy - lastObservedVY * vx, lastObservedVX * vx + lastObservedVY * vy) / dt;
                if (Math.Abs(turn) > .16f && Math.Abs(turn) < 3.2f)
                {
                    float sign = Math.Sign(turn);
                    if (sign != turnSign) { turnHistory = 0; CircleConfidence *= .2f; }
                    turnSign = sign; turnHistory = Math.Min(3, turnHistory + dt);
                    ObservedTurnRate += (turn - ObservedTurnRate) * Math.Min(1, dt * 5);
                }
                else
                {
                    turnHistory = Math.Max(0, turnHistory - dt * 2);
                    ObservedTurnRate *= Math.Max(0, 1 - dt * 5);
                    if (Math.Abs(turn) >= 3.2f) CircleConfidence *= .25f;
                }
            }
            else { turnHistory = Math.Max(0, turnHistory - dt); ObservedTurnRate *= Math.Max(0, 1 - dt * 3); }
            CircleConfidence += (Clamp((turnHistory - .35f) / 1.1f, 0, 1) - CircleConfidence) * Math.Min(1, dt * 3);
            lastObservedVX = vx; lastObservedVY = vy;
            if (speed > 300) { vx *= 300 / speed; vy *= 300 / speed; }
            float blend = Math.Min(1, dt * 12);
            PlayerVX += (vx - PlayerVX) * blend; PlayerVY += (vy - PlayerVY) * blend;
            PredictPosition(1.2f, out PredictedX, out PredictedY);
            MotionReadout = ObservedDashAge < .6f ? "检测到冲刺 · 截击者准备接力" : CircleConfidence > .55f ? "弧线轨迹已识别 · 变向或切入内圈" : "线性推演 · 蓄力后变向可欺骗瞄准";
        }

        private void PredictPosition(float seconds, out float tx, out float ty)
        {
            if (DecoyTime > 0) { tx = DecoyX; ty = DecoyY; return; }
            seconds = Clamp(seconds, 0, 2.6f);
            float omega = CircleConfidence > .35f ? ObservedTurnRate * CircleConfidence : 0;
            if (Math.Abs(omega) > .12f)
            {
                // Integrate the observed turn rate, rather than aiming along an impossible tangent.
                float angle = omega * seconds, sin = (float)Math.Sin(angle), cos = (float)Math.Cos(angle);
                tx = X + (PlayerVX * sin + PlayerVY * (cos - 1)) / omega;
                ty = Y + (PlayerVX * (1 - cos) + PlayerVY * sin) / omega;
            }
            else { tx = X + PlayerVX * seconds; ty = Y + PlayerVY * seconds; }
            tx = Clamp(tx, Left + 18, Right - 18); ty = Clamp(ty, Top + 18, Bottom - 18);
        }

        private Enemy FindEnemy(int id)
        {
            if (id <= 0) return null;
            Enemy found;
            if (tacticalIndex.TryGetValue(id, out found)) return found;
            for (int i = 0; i < Enemies.Count; i++) if (Enemies[i].Id == id) return Enemies[i];
            return null;
        }

        private void SpawnSquad()
        {
            if (Enemies.Count > EnemyLimit - 9) return;
            int relayCount = 0;
            for (int i = 0; i < Enemies.Count; i++) if (Enemies[i].Kind == "relay" && Enemies[i].Hp > 0) relayCount++;
            if (relayCount >= 3) return;
            Spawn("relay"); Enemy leader = Enemies[Enemies.Count - 1];
            string[] kinds = Time >= 90 ? new string[] { "tank", "shooter", "shooter", "runner", "runner", "bomber", "repairer", "prism" }
                : new string[] { "tank", "shooter", "shooter", "runner", "runner", "bomber" };
            for (int i = 0; i < kinds.Length; i++)
            {
                Spawn(kinds[i]); Enemy member = Enemies[Enemies.Count - 1];
                float angle = (float)(i * Math.PI * 2 / kinds.Length);
                member.X = Clamp(leader.X + (float)Math.Cos(angle) * 82, Left + member.R, Right - member.R);
                member.Y = Clamp(leader.Y + (float)Math.Sin(angle) * 82, Top + member.R, Bottom - member.R);
                member.RelayId = leader.Id;
                member.Cooldown += i * .28f;
                // Spawn graphics must reflect the actual squad position.
                AddEffect("spawn", member.X, member.Y, 0, 0, member.R + 8, .55f);
            }
            // A squad replaces part of the normal drip instead of adding unbounded density.
            spawnClock += 1.4f;
            tacticClock = 0;
        }

        private void UpdateSquadPhase(Enemy relay, float dt)
        {
            if (relay.Recovery > 0 || relay.JamTime > 0 || relay.Stun > 0)
            { relay.SquadClock = 0; relay.BrainState = "恢复"; relay.PhaseLeft = Math.Max(relay.Recovery, relay.JamTime); return; }
            if (relay.AttackPhase == -1) { relay.BrainState = "准备"; relay.PhaseLeft = 1.15f; return; }
            relay.SquadClock = (relay.SquadClock + dt) % 6.9f;
            float clock = relay.SquadClock;
            relay.BrainState = clock < 1.15f ? "准备" : clock < 3.05f ? "压制" : clock < 5.25f ? "截击" : "恢复";
            relay.PhaseLeft = (clock < 1.15f ? 1.15f : clock < 3.05f ? 3.05f : clock < 5.25f ? 5.25f : 6.9f) - clock;
            relay.Intent = relay.BrainState == "准备" ? "展开交叉火力" : relay.BrainState == "压制" ? "远程封锁路线" : relay.BrainState == "截击" ? "近战接力围堵" : "重组 · 节点暴露";
        }

        private void UpdateTactics(float dt)
        {
            tacticalIndex.Clear();
            for (int i = 0; i < Enemies.Count; i++) tacticalIndex[Enemies[i].Id] = Enemies[i];
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i];
                if (e.Kind == "relay" && e.Hp <= 0) DisruptRelay(e);
                e.Recovery = Math.Max(0, e.Recovery - dt);
                if (e.Kind == "relay" && e.Hp > 0) UpdateSquadPhase(e, dt);
                if (e.RelayId != 0)
                {
                    Enemy relay = FindEnemy(e.RelayId);
                    if (relay == null || relay.Hp <= 0 || relay.JamTime > 0 || relay.Stun > 0 || relay.Recovery > 0 || DistanceSquared(e.X, e.Y, relay.X, relay.Y) > 460 * 460) e.RelayId = 0;
                }
                e.Shielded = false;
            }
            tacticClock -= dt;
            if (tacticClock <= 0)
            {
                tacticClock = .22f;
                AssignSquads(); AssignGuards();
                for (int i = 0; i < Enemies.Count; i++) PlanPosition(Enemies[i]);
            }
            bool coordinated = false; Enemy visibleLeader = null; float leaderDistance = float.MaxValue;
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i]; if (e.Hp <= 0) continue;
                e.Shielded = IsProtected(e);
                if (e.RelayId > 0) coordinated = true;
                if (e.Kind == "relay")
                {
                    float distance = DistanceSquared(X, Y, e.X, e.Y);
                    if (distance < leaderDistance) { leaderDistance = distance; visibleLeader = e; }
                }
                else if (e.RelayId > 0)
                {
                    Enemy leader = FindEnemy(e.RelayId); if (leader != null) { e.BrainState = leader.BrainState; e.PhaseLeft = leader.PhaseLeft; }
                }
            }
            Enemy focused = FindEnemy(FocusTargetId);
            if (focused == null || focused.Hp <= 0) FocusTargetId = 0;
            SquadPhase = visibleLeader != null ? visibleLeader.BrainState : "游猎";
            SquadPhaseLeft = visibleLeader != null ? visibleLeader.PhaseLeft : 0;
            TacticName = coordinated ? "中继合围 · " + SquadPhase + " · E 锁节点，切断下一轮接力" : "游猎 · " + MotionReadout;
        }

        private void AssignSquads()
        {
            Dictionary<int, int> assigned = new Dictionary<int, int>();
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i]; e.RelayId = 0;
                if (e.Kind == "relay" || e.Kind == "boss" || e.Hp <= 0 || e.Recovery > 0 || e.JamTime > 0) continue;
                float best = 420 * 420;
                for (int j = 0; j < Enemies.Count; j++)
                {
                    Enemy relay = Enemies[j];
                    if (relay.Kind != "relay" || relay.Hp <= 0 || relay.AttackPhase == -1 || relay.JamTime > 0 || relay.Stun > 0 || relay.Recovery > 0) continue;
                    int members; if (assigned.TryGetValue(relay.Id, out members) && members >= 10) continue;
                    float d = DistanceSquared(e.X, e.Y, relay.X, relay.Y);
                    if (d < best) { best = d; e.RelayId = relay.Id; }
                }
                if (e.RelayId > 0) { int members; assigned.TryGetValue(e.RelayId, out members); assigned[e.RelayId] = members + 1; }
            }
        }

        private void AssignGuards()
        {
            for (int i = 0; i < Enemies.Count; i++) Enemies[i].GuardId = 0;
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy guard = Enemies[i];
                if (guard.Kind != "tank" || guard.Hp <= 0 || guard.JamTime > 0 || guard.Stun > 0 || guard.Recovery > 0) continue;
                Enemy target = null; float best = 235 * 235;
                for (int j = 0; j < Enemies.Count; j++)
                {
                    Enemy e = Enemies[j];
                    if ((e.Kind != "shooter" && e.Kind != "relay" && e.Kind != "repairer" && e.Kind != "prism") || e.Hp <= 0 || e.GuardId != 0 || e.Stun > 0) continue;
                    if (guard.RelayId > 0 && e.RelayId != guard.RelayId && e.Id != guard.RelayId) continue;
                    float distance = DistanceSquared(guard.X, guard.Y, e.X, e.Y);
                    if (e.Kind == "relay") distance *= 1.3f;
                    if (e.Kind == "prism") distance *= .65f;
                    if (distance < best) { best = distance; target = e; }
                }
                if (target != null) { guard.GuardId = target.Id; target.GuardId = guard.Id; }
            }
        }

        private void PlanPosition(Enemy e)
        {
            float tx = DecoyTime > 0 ? DecoyX : X, ty = DecoyTime > 0 ? DecoyY : Y;
            float dx = e.X - tx, dy = e.Y - ty, distance = (float)Math.Sqrt(dx * dx + dy * dy);
            float nx = dx / Math.Max(1, distance), ny = dy / Math.Max(1, distance);
            float side = (e.Id & 1) == 0 ? 1 : -1;
            bool linked = e.RelayId > 0 && e.Recovery <= 0;
            e.OrderX = tx; e.OrderY = ty; e.Order = "追踪";
            e.Role = e.Kind == "relay" ? "指挥" : e.Kind == "repairer" ? "续航" : e.Kind == "prism" ? "破阵" : e.Kind == "tank" ? "掩护" : e.Kind == "runner" ? "截击" : e.Kind == "shooter" ? "压制" : e.Kind == "bomber" ? "封路" : "驱赶";
            if (e.Kind == "tank" && e.GuardId > 0)
            {
                Enemy ally = FindEnemy(e.GuardId);
                if (ally != null)
                {
                    float ax = tx - ally.X, ay = ty - ally.Y, ad = (float)Math.Sqrt(ax * ax + ay * ay);
                    e.OrderX = ally.X + ax / Math.Max(ad, 1) * 62;
                    e.OrderY = ally.Y + ay / Math.Max(ad, 1) * 62;
                    e.Order = "掩护";
                }
            }
            else if (e.Kind == "repairer")
            {
                Enemy patient = FindRepairTarget(e);
                if (patient != null)
                {
                    float ax = patient.X - tx, ay = patient.Y - ty, ad = (float)Math.Sqrt(ax * ax + ay * ay);
                    e.OrderX = patient.X + ax / Math.Max(1, ad) * 75; e.OrderY = patient.Y + ay / Math.Max(1, ad) * 75;
                    e.Order = "搜寻伤员";
                }
                else { e.OrderX = tx + nx * 355 - ny * side * 70; e.OrderY = ty + ny * 355 + nx * side * 70; e.Order = "后方待命"; }
            }
            else if (e.Kind == "shooter" || e.Kind == "relay" || e.Kind == "bomber")
            {
                float orbit = e.Kind == "relay" ? 345 : e.Kind == "bomber" ? 325 : 300;
                Enemy leader = FindEnemy(e.RelayId);
                if (linked && leader != null)
                {
                    dx = leader.X - tx; dy = leader.Y - ty; distance = (float)Math.Sqrt(dx * dx + dy * dy);
                    nx = dx / Math.Max(distance, 1); ny = dy / Math.Max(distance, 1);
                }
                float lateral = linked && e.Kind == "shooter" ? side * 145 : side * 50;
                e.OrderX = tx + nx * orbit - ny * lateral; e.OrderY = ty + ny * orbit + nx * lateral;
                e.Order = e.Kind == "relay" ? "指挥" : e.Kind == "bomber" ? "封路" : linked ? "交叉火力" : "拉开距离";
            }
            else if ((e.Kind == "chaser" || e.Kind == "runner" || e.Kind == "prism") && distance > 75)
            {
                float predict = linked ? 1.15f : .55f;
                float lateral = linked ? 80 : 28;
                PredictPosition(predict, out e.OrderX, out e.OrderY);
                e.OrderX -= ny * side * lateral; e.OrderY += nx * side * lateral;
                e.Order = linked ? "侧翼截击" : "弧线切入";
                Enemy leader = FindEnemy(e.RelayId);
                if (linked && leader != null && (leader.BrainState == "准备" || leader.BrainState == "压制"))
                {
                    // Hold both lateral approach routes while ranged units force a response.
                    e.OrderX += nx * 145; e.OrderY += ny * 145; e.Order = "保留截击";
                }
            }
            if (e.Kind != "relay" && e.AttackPhase == 0 && e.Recovery <= 0 && e.Channel <= 0) e.Intent = e.Order;
            e.OrderX = Clamp(e.OrderX, Left + e.R, Right - e.R);
            e.OrderY = Clamp(e.OrderY, Top + e.R, Bottom - e.R);
        }

        private Enemy FindRepairTarget(Enemy healer)
        {
            Enemy target = null; float best = float.MaxValue;
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i];
                if (e == healer || e.Kind == "boss" || e.Kind == "repairer" || e.Hp <= 0 || e.AttackPhase == -1 || e.Hp >= e.MaxHp * .86f) continue;
                float distance = DistanceSquared(e.X, e.Y, healer.X, healer.Y);
                if (distance > 260 * 260) continue;
                float score = distance * (e.Kind == "relay" ? .4f : e.Kind == "prism" ? .7f : 1) + (e.Hp / Math.Max(e.MaxHp, 1)) * 3000;
                if (score < best) { target = e; best = score; }
            }
            return target;
        }

        private void StopRepair(Enemy e)
        {
            e.Channel = 0; e.RepairTargetId = 0; e.AttackPhase = 0; e.Windup = 0;
            e.Recovery = Math.Max(e.Recovery, 1.8f); e.Cooldown = Math.Max(e.Cooldown, 3.1f);
            e.BrainState = "恢复"; e.Intent = "修复被打断";
        }

        private bool UpdateRepair(Enemy e, float dt)
        {
            if (e.Channel > 0)
            {
                Enemy patient = FindEnemy(e.RepairTargetId);
                if (e.JamTime > 0 || e.Stun > 0 || e.Recovery > 0 || patient == null || patient.Hp <= 0 || DistanceSquared(e.X, e.Y, patient.X, patient.Y) > 280 * 280)
                { StopRepair(e); return false; }
                e.Channel = Math.Max(0, e.Channel - dt); e.VX = e.VY = 0; e.BrainState = "引导"; e.Intent = "修复 " + EnemyName(patient.Kind);
                if (e.Channel <= 0)
                {
                    // A bounded pulse, never resurrection or permanent damage negation.
                    patient.Hp = Math.Min(patient.MaxHp, patient.Hp + Math.Min(patient.MaxHp * .18f, 35 + Math.Min(Time, 900) * .06f));
                    e.AttackSerial++;
                    AddEffect("repair", e.X, e.Y, patient.X, patient.Y, 12, .5f);
                    e.RepairTargetId = 0; e.Cooldown = 5.2f; e.Recovery = .65f;
                }
                return true;
            }
            if (e.Cooldown <= 0 && e.Recovery <= 0 && e.JamTime <= 0 && e.BurnTime <= 0 && Phase != "喘息")
            {
                Enemy patient = FindRepairTarget(e);
                if (patient != null && DistanceSquared(e.X, e.Y, patient.X, patient.Y) < 220 * 220)
                { e.RepairTargetId = patient.Id; e.Channel = e.MaxChannel = 1.6f; e.Intent = "引导修复 · 命中可打断"; return true; }
            }
            return false;
        }

        private void UpdateFacing(Enemy e, float dt)
        {
            if (e.AttackPhase == 1 || e.AttackPhase == 2 || e.Stun > 0) return;
            float angle = (float)Math.Atan2(e.FacingY, e.FacingX), target = (float)Math.Atan2(Y - e.Y, X - e.X);
            float difference = (float)Math.Atan2(Math.Sin(target - angle), Math.Cos(target - angle));
            angle += Clamp(difference, -1.65f * dt, 1.65f * dt);
            e.FacingX = (float)Math.Cos(angle); e.FacingY = (float)Math.Sin(angle);
        }

        private float ApplyTacticalDamage(Enemy e, float damage, float sourceX, float sourceY)
        {
            if (e.Kind == "repairer" && e.Channel > 0 && damage >= 5) StopRepair(e);
            if (e.Kind == "boss" && e.Recovery > 0) damage *= 1.35f;
            if (e.Kind == "prism" && e.JamTime <= 0 && e.Stun <= 0 && e.Recovery <= 0)
            {
                float dx = sourceX - e.X, dy = sourceY - e.Y, length = (float)Math.Sqrt(dx * dx + dy * dy);
                float dot = (dx * e.FacingX + dy * e.FacingY) / Math.Max(.01f, length);
                if (dot > .5f) damage *= .35f;
                else
                {
                    damage *= dot < -.35f ? 1.45f : 1.10f;
                    if (damage >= 8 && e.AttackPhase == 1)
                    { e.AttackPhase = 0; e.Windup = 0; e.Recovery = 1.1f; e.Cooldown = 2.7f; CancelEnemyWarning(e.Id); e.Intent = "护面被击穿"; }
                }
            }
            return damage;
        }

        private bool IsProtected(Enemy e)
        {
            if (e.Hp <= 0 || e.GuardId <= 0 || (e.Kind != "shooter" && e.Kind != "relay" && e.Kind != "repairer" && e.Kind != "prism")) return false;
            Enemy guard = FindEnemy(e.GuardId);
            if (guard == null || guard.Kind != "tank" || guard.Hp <= 0 || guard.Stun > 0 || guard.JamTime > 0 || guard.Recovery > 0 || guard.GuardId != e.Id) return false;
            float dx = X - e.X, dy = Y - e.Y, length = (float)Math.Sqrt(dx * dx + dy * dy);
            float gx = guard.X - e.X, gy = guard.Y - e.Y;
            float forward = (gx * dx + gy * dy) / Math.Max(1, length);
            float lateral = Math.Abs(gx * dy - gy * dx) / Math.Max(1, length);
            // Shield works only while the guard physically occupies the firing lane.
            return forward > 12 && forward < length && lateral < 44 && gx * gx + gy * gy < 125 * 125;
        }

        public void CancelEnemyWarning(int id)
        {
            for (int i = Hazards.Count - 1; i >= 0; i--) if (Hazards[i].OwnerId == id && Hazards[i].Warning > 0) RemoveFast(Hazards, i);
        }

        public void DisruptRelay(Enemy relay)
        {
            if (relay == null || relay.Kind != "relay" || relay.RelayBroken) return;
            if (relay.Hp <= 0) relay.RelayBroken = true;
            relay.Recovery = Math.Max(relay.Recovery, 3.2f);
            int count = 0;
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i];
                if (e.Hp <= 0 || e.Kind == "boss" || (e.RelayId != relay.Id && e.GuardId != relay.Id)) continue;
                e.RelayId = e.GuardId = 0; e.Shielded = false;
                e.Channel = 0; e.RepairTargetId = 0;
                e.Stun = Math.Max(e.Stun, 1.4f); e.Recovery = Math.Max(e.Recovery, 3.2f);
                e.AttackPhase = 0; e.Windup = 0; e.Cooldown = Math.Max(e.Cooldown, 3.0f);
                e.VX = e.VY = 0; e.Order = "链路中断"; CancelEnemyWarning(e.Id); count++;
            }
            if (count > 0)
            {
                AddEffect("disrupt", relay.X, relay.Y, 0, 0, 150, .65f);
                Say("指挥链中断 · 编队失序，趁机突破", 2.8f); Event("break");
            }
            tacticClock = 0;
        }

        private void UpdateTacticalEnemies(float dt)
        {
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i]; if (e.Hp <= 0) continue;
                e.Life += dt;
                if (e.AttackPhase == -1) { e.Timer -= dt; if (e.Timer <= 0) { e.AttackPhase = 0; PlanPosition(e); } continue; }
                UpdateFacing(e, dt);
                if (e.Kind == "boss")
                {
                    e.BossStage = e.Hp < e.MaxHp * .5f ? 2 : 1;
                    e.BrainState = e.Recovery > 0 ? "恢复" : e.AttackPhase == 1 ? "蓄力" : "追猎";
                }
                if (e.Kind == "repairer" && e.Channel > 0 && (e.Stun > 0 || e.JamTime > 0 || e.Recovery > 0)) StopRepair(e);
                if (e.Stun > 0) { e.Stun = Math.Max(0, e.Stun - dt); e.VX = e.VY = 0; continue; }
                e.Cooldown -= dt;
                if (e.Kind == "repairer" && UpdateRepair(e, dt)) continue;
                if (e.AttackPhase == 1)
                {
                    e.VX = e.VY = 0; e.Windup -= dt;
                    if (e.Windup <= 0) FinishAttack(e);
                }
                else if (e.AttackPhase == 2)
                {
                    e.X += e.VX * dt; e.Y += e.VY * dt; e.Timer -= dt;
                    if (e.Timer <= 0) { e.AttackPhase = 0; e.Cooldown = 2.8f; e.Recovery = Math.Max(e.Recovery, .75f); e.Intent = "截击落空 · 暴露"; }
                }
                else
                {
                    float distance = DistanceSquared(X, Y, e.X, e.Y);
                    bool special = e.Kind == "runner" || e.Kind == "shooter" || e.Kind == "bomber" || e.Kind == "boss" || e.Kind == "prism";
                    float range = e.Kind == "runner" ? 450 : e.Kind == "prism" ? 440 : e.Kind == "boss" ? 1100 : 630;
                    if (special && e.Cooldown <= 0 && e.JamTime <= 0 && e.Recovery <= 0 && distance < range * range && TryBeginAttack(e)) { }
                    else
                    {
                        float dx = e.OrderX - e.X, dy = e.OrderY - e.Y, d = (float)Math.Sqrt(dx * dx + dy * dy);
                        float speed = e.Speed * (e.JamTime > 0 ? .5f : 1);
                        float travel = Math.Min(d, speed * dt);
                        if (e.Recovery > 0 && e.Kind == "runner") travel *= .35f;
                        if (e.Recovery > 0 && (e.Kind == "boss" || e.Kind == "prism")) travel *= .18f;
                        e.VX = d > 6 ? dx / Math.Max(d, 1) * travel / dt : 0;
                        e.VY = d > 6 ? dy / Math.Max(d, 1) * travel / dt : 0;
                        e.X += e.VX * dt; e.Y += e.VY * dt;
                    }
                }
                e.X = Clamp(e.X, Left + e.R, Right - e.R); e.Y = Clamp(e.Y, Top + e.R, Bottom - e.R);
                if (DistanceSquared(X, Y, e.X, e.Y) < (e.R + 11) * (e.R + 11)) Hurt(e.Damage);
                if (Mode == "over") return;
            }
        }

        private bool TryBeginAttack(Enemy e)
        {
            if (Phase == "喘息" && e.Kind != "boss") return false;
            Enemy leader = FindEnemy(e.RelayId);
            if (leader != null)
            {
                if ((e.Kind == "shooter" || e.Kind == "bomber") && leader.BrainState != "压制") return false;
                if ((e.Kind == "runner" || e.Kind == "prism") && leader.BrainState != "截击") return false;
            }
            if (Time < nextSpecial) return false;
            int active = 0;
            bool bossWaiting = false;
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy other = Enemies[i]; if (other.Hp <= 0) continue;
                if (other.AttackPhase == 1 || other.AttackPhase == 2) active++;
                if (other.Kind == "boss" && other.AttackPhase == 0 && other.Cooldown <= 0) bossWaiting = true;
            }
            if (active >= SpecialAttackLimit) return false;
            if (e.Kind != "boss" && bossWaiting && active >= SpecialAttackLimit - 1) return false;
            bool area = e.Kind == "bomber" || e.Kind == "boss" || e.Kind == "prism";
            if (area && Hazards.Count >= 5) return false;
            float tx = DecoyTime > 0 ? DecoyX : X, ty = DecoyTime > 0 ? DecoyY : Y;
            float lead = e.Kind == "runner" ? 1.10f : e.Kind == "shooter" ? 1.70f : e.Kind == "prism" ? .80f : e.Kind == "boss" ? 1.10f : 1.20f;
            if (e.Kind == "shooter" && DecoyTime <= 0)
            {
                // Solve an intercept using the observed velocity and a fixed 0.9s firing tell.
                float px = X + PlayerVX * .9f - e.X, py = Y + PlayerVY * .9f - e.Y;
                float a = PlayerVX * PlayerVX + PlayerVY * PlayerVY - 255 * 255;
                float b = 2 * (px * PlayerVX + py * PlayerVY), c = px * px + py * py;
                float discriminant = b * b - 4 * a * c;
                if (discriminant >= 0 && Math.Abs(a) > .01f)
                {
                    float t1 = (-b - (float)Math.Sqrt(discriminant)) / (2 * a), t2 = (-b + (float)Math.Sqrt(discriminant)) / (2 * a);
                    float travel = t1 > 0 ? t1 : t2 > 0 ? t2 : .8f;
                    lead = Math.Min(2.6f, .9f + travel);
                }
            }
            if (e.RelayId > 0 && e.Kind != "shooter") lead += .05f;
            if (e.Kind == "runner" && ObservedDashAge > .20f && ObservedDashAge < 1.05f) lead = .9f;
            if (e.Kind == "boss" && e.AttackSerial % 3 == 2) lead = .35f;
            PredictPosition(lead, out tx, out ty);
            float radius = e.Kind == "boss" ? 96 : 64;
            tx = Clamp(tx, Left + 18, Right - 18); ty = Clamp(ty, Top + 18, Bottom - 18);
            float windup = e.Kind == "runner" ? .78f : e.Kind == "shooter" ? .9f : e.Kind == "prism" ? .85f : e.Kind == "boss" ? 1.15f : 1.25f;
            List<Hazard> proposed = area ? AttackHazards(e, tx, ty, windup) : null;
            if (!LeavesEscape(e, tx, ty, area ? radius : 22, proposed)) { e.Cooldown = .28f; return false; }
            e.AttackPhase = 1; e.TargetX = tx; e.TargetY = ty;
            e.Windup = windup;
            e.Intent = e.Kind == "boss" ? (e.AttackSerial % 3 == 0 ? "审判圆区" : e.AttackSerial % 3 == 1 ? "棱线切割" : "回声环震") : e.Kind == "prism" ? "护面突刺 · 侧背可打断" : e.Kind == "runner" && ObservedDashAge < 1.05f ? "冲刺后接力" : "锁定 · 不再追踪";
            if (e.Kind == "boss") e.BrainState = "蓄力";
            e.VX = e.VY = 0;
            // Stagger the starts: no frame can suddenly open every attack lane.
            nextSpecial = Time + .30f;
            if (area)
            {
                for (int i = 0; i < proposed.Count && Hazards.Count < HazardLimit; i++) Hazards.Add(proposed[i]);
            }
            Event("warning"); return true;
        }

        private List<Hazard> AttackHazards(Enemy e, float tx, float ty, float warning)
        {
            List<Hazard> result = new List<Hazard>(2);
            Hazard h = new Hazard { OwnerId = e.Id, X = tx, Y = ty, R = e.Kind == "boss" ? 96 : 64,
                Warning = warning, Life = e.Kind == "boss" ? 1.0f : 2.0f, Damage = e.Damage * .80f };
            if (e.Kind == "prism")
            {
                h.Shape = "line"; h.X = e.X; h.Y = e.Y;
                h.X2 = Clamp(e.X + e.FacingX * 420, Left + 18, Right - 18);
                h.Y2 = Clamp(e.Y + e.FacingY * 420, Top + 18, Bottom - 18);
                h.R = 28; h.Life = .28f;
            }
            else if (e.Kind == "boss" && e.AttackSerial % 3 == 1)
            {
                float dx = PlayerVX, dy = PlayerVY, distance = (float)Math.Sqrt(dx * dx + dy * dy);
                if (distance < 30) { dx = tx - e.X; dy = ty - e.Y; distance = (float)Math.Sqrt(dx * dx + dy * dy); }
                dx /= Math.Max(1, distance); dy /= Math.Max(1, distance);
                if (Math.Abs(dx) + Math.Abs(dy) < .01f) dx = 1;
                h.Shape = "line"; h.R = e.BossStage >= 2 ? 24 : 28; h.Life = .8f;
                h.X = Clamp(tx - dy * 220, Left + 18, Right - 18); h.Y = Clamp(ty + dx * 220, Top + 18, Bottom - 18);
                h.X2 = Clamp(tx + dy * 220, Left + 18, Right - 18); h.Y2 = Clamp(ty - dx * 220, Top + 18, Bottom - 18);
                if (e.BossStage >= 2)
                {
                    Hazard second = new Hazard { OwnerId = e.Id, Shape = "line", Warning = warning, Life = h.Life, Damage = h.Damage, R = 24,
                        X = Clamp(h.X + dx * 130, Left + 18, Right - 18), Y = Clamp(h.Y + dy * 130, Top + 18, Bottom - 18),
                        X2 = Clamp(h.X2 + dx * 130, Left + 18, Right - 18), Y2 = Clamp(h.Y2 + dy * 130, Top + 18, Bottom - 18) };
                    result.Add(second);
                }
            }
            else if (e.Kind == "boss" && e.AttackSerial % 3 == 2)
            { h.Shape = "ring"; h.R = 180; h.InnerR = 100; h.Life = .9f; }
            result.Add(h); return result;
        }

        public static bool HazardContains(Hazard h, float px, float py, float padding)
        {
            if (h.Shape == "line")
                return WorldSegmentDistanceSquared(px, py, h.X, h.Y, h.X2, h.Y2) < (h.R + padding) * (h.R + padding);
            float distance = DistanceSquared(px, py, h.X, h.Y);
            if (h.Shape == "ring") return distance < (h.R + padding) * (h.R + padding) && distance > Math.Max(0, h.InnerR - padding) * Math.Max(0, h.InnerR - padding);
            return distance < (h.R + padding) * (h.R + padding);
        }

        private bool LeavesEscape(Enemy attacker, float tx, float ty, float radius, List<Hazard> proposed)
        {
            // Reserve at least one walkable direction. A warning is not yet a blocking floor:
            // allow crossing it before detonation, but always require a safe destination.
            // This is a bounded eight-sector fairness check, not navigation over the whole arena.
            bool proposedArea = proposed != null;
            float proposedWarning = proposedArea && proposed.Count > 0 ? proposed[0].Warning : 0;
            // Staying inside a ring or taking a short step is also a valid response.
            for (int route = 0; route < 3; route++) for (int sector = 0; sector < (route == 0 ? 1 : 8); sector++)
            {
                float travel = route == 0 ? 0 : route == 1 ? 80 : 175;
                float angle = sector * (float)Math.PI / 4;
                float ex = X + (float)Math.Cos(angle) * travel, ey = Y + (float)Math.Sin(angle) * travel;
                if (ex < Left + 18 || ex > Right - 18 || ey < Top + 18 || ey > Bottom - 18) continue;
                bool safe = true;
                for (int sample = 1; sample <= 2 && safe; sample++)
                {
                    float px = X + (ex - X) * sample / 2, py = Y + (ey - Y) * sample / 2;
                    float arrive = travel * sample / 2 / (220 + Rank("speed") * 16);
                    if (!proposedArea && AttackCovers(attacker, tx, ty, radius, px, py)) { safe = false; break; }
                    if (proposedArea && (sample == 2 || proposedWarning <= arrive))
                        for (int j = 0; j < proposed.Count && safe; j++) if (HazardContains(proposed[j], px, py, 18)) safe = false;
                    for (int j = 0; j < Hazards.Count && safe; j++)
                        if ((sample == 2 || Hazards[j].Warning <= arrive) && HazardContains(Hazards[j], px, py, 18)) safe = false;
                    for (int j = 0; j < Enemies.Count && safe; j++)
                    {
                        Enemy e = Enemies[j];
                        if (e.Hp <= 0 || e.AttackPhase != 1 || e.Kind == "boss" || e.Kind == "bomber" || e.Kind == "prism") continue;
                        if (AttackCovers(e, e.TargetX, e.TargetY, 24, px, py)) safe = false;
                    }
                }
                if (safe) return true;
            }
            return false;
        }

        private bool AttackCovers(Enemy e, float tx, float ty, float radius, float px, float py)
        {
            if (e.Kind == "bomber" || e.Kind == "boss") return DistanceSquared(px, py, tx, ty) < (radius + 18) * (radius + 18);
            float dx = tx - e.X, dy = ty - e.Y, d2 = dx * dx + dy * dy;
            float t = Clamp(((px - e.X) * dx + (py - e.Y) * dy) / Math.Max(1, d2), 0, 1.4f);
            return DistanceSquared(px, py, e.X + dx * t, e.Y + dy * t) < (radius + 15) * (radius + 15);
        }

        private void FinishAttack(Enemy e)
        {
            e.AttackSerial++;
            if (e.Kind == "runner")
            {
                float dx = e.TargetX - e.X, dy = e.TargetY - e.Y, d = (float)Math.Sqrt(dx * dx + dy * dy);
                e.VX = dx / Math.Max(1, d) * 520; e.VY = dy / Math.Max(1, d) * 520;
                e.AttackPhase = 2; e.Timer = Clamp(d / 520, .35f, .86f);
            }
            else if (e.Kind == "shooter")
            {
                float angle = (float)Math.Atan2(e.TargetY - e.Y, e.TargetX - e.X);
                int count = Time > 180 ? 3 : 1;
                for (int j = 0; j < count; j++) AddShot(e.X, e.Y, angle + (j - (count - 1) / 2f) * .18f, 255, e.Damage, true, 0, "enemy");
                e.AttackPhase = 0; e.Cooldown = e.RelayId > 0 ? 2.4f : 2.9f; e.Recovery = .32f;
            }
            else
            {
                bool resolvedHazard = false;
                for (int i = 0; i < Hazards.Count; i++)
                {
                    Hazard h = Hazards[i]; if (h.OwnerId != e.Id || h.Warning <= 0) continue;
                    resolvedHazard = true;
                    h.Warning = 0;
                    if (h.Shape == "circle") AddEffect("aoeBlast", h.X, h.Y, 0, 0, h.R, .42f);
                    if (HazardContains(h, X, Y, 10)) Hurt(e.Damage * 1.25f);
                }
                // Public scenario fixtures can supply a legacy circular windup without a Hazard.
                if (!resolvedHazard && (e.Kind == "bomber" || (e.Kind == "boss" && (e.AttackSerial - 1) % 3 == 0)))
                {
                    float radius = e.Kind == "boss" ? 96 : 64;
                    AddEffect("aoeBlast", e.TargetX, e.TargetY, 0, 0, radius, .42f);
                    if (DistanceSquared(X, Y, e.TargetX, e.TargetY) < (radius + 10) * (radius + 10)) Hurt(e.Damage * 1.25f);
                }
                if (e.Kind == "boss")
                {
                    // Every distinct attack ends in a visible punish window; no simultaneous bullet ring.
                    e.Recovery = e.BossStage >= 2 ? 1.45f : 1.8f; e.BrainState = "恢复"; e.Intent = "核心暴露 · 受到伤害 +35%";
                }
                if (e.Kind == "prism") { e.Recovery = 1.25f; e.Intent = "护甲过热 · 反击窗口"; }
                e.AttackPhase = 0; e.Cooldown = e.Kind == "boss" ? (e.BossStage >= 2 ? 2.55f : 3.0f) : e.Kind == "prism" ? 3.5f : 3.8f;
            }
        }

        private void UpdateHazards(float dt)
        {
            for (int i = Hazards.Count - 1; i >= 0; i--)
            {
                Hazard h = Hazards[i];
                if (h.Warning > 0)
                {
                    Enemy owner = FindEnemy(h.OwnerId);
                    if (owner == null || owner.Hp <= 0 || owner.AttackPhase != 1) { RemoveFast(Hazards, i); continue; }
                    h.Warning = Math.Max(.001f, owner.Windup); continue;
                }
                h.Life -= dt;
                if (h.Life <= 0) { RemoveFast(Hazards, i); continue; }
                if (HazardContains(h, X, Y, 8)) Hurt(h.Damage);
                if (Mode == "over") return;
            }
        }
    }

            DecoyX = lastWorldX = X; DecoyY = lastWorldY = Y; worldDashSerial = 0;

            Ranks[starter] = 1;
            if (starter == "forge_ward") Ward = 14;
        }

        private void UpdateWorld(float dt, bool dashedNow)
        {
            wardWait = Math.Max(0, wardWait - dt);
            wallCooldown = Math.Max(0, wallCooldown - dt);
            interceptClock = Math.Max(0, interceptClock - dt);
            float previousDecoyTime = DecoyTime;
            DecoyTime = Math.Max(0, DecoyTime - dt);
            bool hotEcho = HasWorldSynergy("ember", "echo");
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i]; if (e.Hp <= 0) continue;
                e.JamTime = Math.Max(0, e.JamTime - dt);
                e.ResonanceTime = Math.Max(0, e.ResonanceTime - dt);
                if (e.BurnTime > 0)
                {
                    float tick = Math.Min(dt, e.BurnTime);
                    float multiplier = hotEcho && e.JamTime > 0 ? 1.5f : 1;
                    e.Hp -= e.BurnDps * tick * multiplier;
                    e.BurnTime = Math.Max(0, e.BurnTime - dt);
                    // Keep a tiny burn marker until collection if this tick was lethal.
                    if (e.Hp <= 0) e.BurnTime = Math.Max(.001f, e.BurnTime);
                }
            }
            if (previousDecoyTime > 0 && DecoyTime <= 0 && HasSchoolCapstone("echo"))
                WorldJamPulse(DecoyX, DecoyY, 130, .9f);
            if (dashedNow)
            {
                worldDashSerial++;
                int decoy = Rank("echo_decoy");
                if (decoy > 0 || HasSchoolCapstone("echo"))
                {
                    DecoyX = lastWorldX; DecoyY = lastWorldY;
                    DecoyTime = decoy > 0 ? .9f + .4f * decoy : 1.2f;
                    AddEffect("echoDecoy", DecoyX, DecoyY, 0, 0, 24, DecoyTime);
                }
                int jammer = Rank("echo_jammer");
                if (jammer > 0) WorldJamPulse(X, Y, 105 + jammer * 15, .55f + jammer * .2f);
            }
            int breaker = Rank("forge_breaker");
            if (breaker > 0 && (dashedNow || DashTime > 0))
            {
                for (int i = 0; i < Enemies.Count; i++)
                {
                    Enemy e = Enemies[i]; if (e.Hp <= 0 || e.AttackPhase == -1 || e.WorldDashHit == worldDashSerial) continue;
                    if (WorldSegmentDistanceSquared(e.X, e.Y, lastWorldX, lastWorldY, X, Y) > (42 + e.R) * (42 + e.R)) continue;
                    e.WorldDashHit = worldDashSerial;
                    JamWorldEnemy(e, .7f + .15f * breaker);
                    DamageEnemy(e, (12 + 12 * breaker) * PowerMultiplier, "forge");
                    if (e.Kind != "boss")
                    {
                        float dx = e.X - X, dy = e.Y - Y;
                        float distance = (float)Math.Sqrt(dx * dx + dy * dy);
                        if (distance < 1) { dx = dashX; dy = dashY; distance = 1; }
                        e.X = Clamp(e.X + dx / distance * 48, Left + e.R, Right - e.R);
                        e.Y = Clamp(e.Y + dy / distance * 48, Top + e.R, Bottom - e.R);
                    }
                    AddEffect("forgeBreak", e.X, e.Y, 0, 0, 40, .3f);
                }
            }
            if (Ward < MaxWard && (wardWait <= 0 || (DecoyTime > 0 && HasWorldSynergy("echo", "forge"))))
                Ward = Math.Min(MaxWard, Ward + 6 * dt);
            int intercept = Rank("forge_intercept");
            if (intercept > 0 && interceptClock <= 0)
            {
                int nearest = -1; float distance = (70 + 10 * intercept) * (70 + 10 * intercept);
                for (int i = 0; i < Shots.Count; i++)
                {
                    if (!Shots[i].Hostile) continue;
                    float d = DistanceSquared(X, Y, Shots[i].X, Shots[i].Y);
                    if (d < distance) { distance = d; nearest = i; }
                }
                if (nearest >= 0)
                {
                    Shot shot = Shots[nearest]; AddEffect("intercept", X, Y, shot.X, shot.Y, 8, .2f);
                    RemoveFast(Shots, nearest); interceptClock = Math.Max(.45f, 1.4f - .25f * intercept); Event("arc");
                }
            }
            lastWorldX = X; lastWorldY = Y;
        }

        private static float WorldSegmentDistanceSquared(float px, float py, float ax, float ay, float bx, float by)
        {
            float dx = bx - ax, dy = by - ay, denom = dx * dx + dy * dy;
            float t = denom <= .001f ? 0 : Clamp(((px - ax) * dx + (py - ay) * dy) / denom, 0, 1);
            return DistanceSquared(px, py, ax + t * dx, ay + t * dy);
        }
        private void WorldJamPulse(float x, float y, float radius, float duration)
        {
            AddEffect("jamPulse", x, y, 0, 0, radius, .38f);
            for (int i = 0; i < Enemies.Count; i++)
            {
                Enemy e = Enemies[i];
                if (e.Hp > 0 && e.AttackPhase != -1 && DistanceSquared(x, y, e.X, e.Y) <= radius * radius)
                    JamWorldEnemy(e, duration);
            }
        }
        private void JamWorldEnemy(Enemy e, float duration)
        {
            if (e.Kind == "boss") { e.JamTime = Math.Max(e.JamTime, .25f); return; }
            e.JamTime = Math.Max(e.JamTime, duration);
            if (e.Kind == "relay") DisruptRelay(e);
            if (e.AttackPhase == 1) { e.AttackPhase = 0; e.Windup = 0; CancelEnemyWarning(e.Id); }
            e.Cooldown = Math.Max(e.Cooldown, duration + .35f);
        }

        private float ApplyWorldDamage(Enemy e, float damage, string weapon)
        {
            int breach = Rank("ember_breach");
            if (breach > 0 && (e.AttackPhase == 1 || e.Recovery > 0 || e.JamTime > 0)) damage *= 1 + .18f * breach;
            bool weaponHit = weapon == "bolt" || weapon == "nova" || weapon == "arc" || weapon == "blade";
            if (!weaponHit) return damage;
            int brand = Rank("ember_brand");
            if (brand > 0)
            {
                e.BurnTime = HasSchoolCapstone("ember") ? 4 : 2;
                e.BurnDps = Math.Max(e.BurnDps, (4 + 3 * brand) * PowerMultiplier);
            }
            string category = weapon == "nova" ? "bolt" : weapon;
            int trace = Rank("echo_trace");
            if (trace > 0 && e.ResonanceTime > 0 && e.LastWorldWeapon != "" && e.LastWorldWeapon != category)
            {
                damage *= 1 + .14f * trace;
                AddEffect("resonance", e.X, e.Y, 0, 0, e.R + 7, .18f);
            }
            e.ResonanceTime = 3; e.LastWorldWeapon = category;
            return damage;
        }

        private float AbsorbWorldDamage(float damage)
        {
            wardWait = 4;
            float absorbed = Math.Min(Ward, Math.Max(0, damage));
            float before = Ward; Ward -= absorbed; damage -= absorbed;
            if (absorbed > 0) AddEffect("wardHit", X, Y, 0, 0, 30, .25f);
            if (before > 0 && Ward <= 0 && HasSchoolCapstone("forge") && wallCooldown <= 0)
            {
                wallCooldown = 8; Invulnerable = Math.Max(Invulnerable, .8f);
                for (int i = Shots.Count - 1; i >= 0; i--)
                    if (Shots[i].Hostile && DistanceSquared(X, Y, Shots[i].X, Shots[i].Y) < 160 * 160)
                    {
                        // Hurt can run inside UpdateShots: defer structural removal to that loop.
                        Shots[i].Life = 0; Shots[i].Damage = 0;
                    }
                AddEffect("wardBurst", X, Y, 0, 0, 160, .45f);
                // The breaking hit consumes the ward; the awakened wall absorbs its overflow.
                damage = 0;
            }
            return Math.Max(0, damage);
        }

        private void OnWorldEnemyDeath(Enemy e)
        {
            if (e.BurnTime > 0)
            {
                if (HasWorldSynergy("ember", "forge")) Ward = Math.Min(MaxWard, Ward + 3);
                int spread = Rank("ember_cascade"), spreadCount = 0, blastCount = 0;
                bool awakened = HasSchoolCapstone("ember");
                for (int i = 0; i < Enemies.Count; i++)
                {
                    Enemy other = Enemies[i];
                    if (other == e || other.Hp <= 0 || DistanceSquared(e.X, e.Y, other.X, other.Y) > 100 * 100) continue;
                    if (spread > 0 && spreadCount < spread + 1)
                    {
                        other.BurnTime = Math.Max(other.BurnTime, 2);
                        other.BurnDps = Math.Max(other.BurnDps, e.BurnDps); spreadCount++;
                    }
                    if (awakened && blastCount < 6) { other.Hp -= e.BurnDps * 1.5f; blastCount++; }
                }
                if (spreadCount > 0 || blastCount > 0) AddEffect("emberBurst", e.X, e.Y, 0, 0, 100, .3f);
            }
            if (e.Kind == "boss")
            {
                CompletedBosses++;
                int chapter = Math.Min(Lore.StoryCount - 1, CompletedBosses);
                if (chapter > StoryChapter)
                {
                    StoryChapter = chapter; LastStoryTitle = Lore.StoryTitle(chapter);
                    Say("档案解封 · " + LastStoryTitle + " · J 阅读", 6); Event("story");
                }
            }
        }

        private void OnWorldUpgrade(string id)
        {
            if (!Lore.IsSchoolUpgrade(id)) return;
            if (id == "forge_ward") Ward = Math.Min(MaxWard, Ward + 14);
            string school = Lore.SchoolOf(id);

                AddEffect("evolve", X, Y, 0, 0, 130, .8f); Event("evolve");
            }
        }
